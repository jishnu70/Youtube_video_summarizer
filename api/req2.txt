// api/requests/main.go

import (
	"errors"
	"log"
	"log/slog"
	"net/http"
	"runtime/debug"

	"github.com/gin-gonic/gin"
	"github.com/redis/go-redis/v9"

	config "github.com/jishnu70/Youtube_video_summarizer/internal/config"
	db "github.com/jishnu70/Youtube_video_summarizer/internal/db"
	"github.com/jishnu70/Youtube_video_summarizer/internal/domain"
	repo "github.com/jishnu70/Youtube_video_summarizer/internal/repo"
	service "github.com/jishnu70/Youtube_video_summarizer/internal/service"
	handler "github.com/jishnu70/Youtube_video_summarizer/requests/handler"
)

func initializeClients(c *config.Config) (*db.MongoDB, *redis.Client) {
	mongoClient, err := db.New(c.MongoURI, c.MongoDatabase)
	if err != nil {
		log.Fatalf("failed to connect to MongoDB: %v", err)
	}

	redisClient := redis.NewClient(&redis.Options{
		Addr:     c.RedisAddr,
		Password: c.RedisPassword,
	})

	return mongoClient, redisClient
}

func initializeRepositories(mongoClient *db.MongoDB, redisClient *redis.Client, c *config.Config) (*repo.SummaryRepo, *repo.TaskRepo, *repo.CacheRepo) {
	summaryRepo, err := repo.NewSummaryRepo(mongoClient, c.SummaryCollection)
	if err != nil {
		log.Fatalf("failed to create summary repo: %v", err)
	}
	taskRepo, err := repo.NewTaskRepo(mongoClient, c.TaskCollection)
	if err != nil {
		log.Fatalf("failed to create task repo: %v", err)
	}
	cacheRepo := repo.NewCacheRepo(redisClient)

	return summaryRepo, taskRepo, cacheRepo
}

func initializeServices(summaryRepo *repo.SummaryRepo, taskRepo *repo.TaskRepo, cacheRepo *repo.CacheRepo) (*service.SummaryService, *service.TaskService, *service.CacheService) {
	cacheService, err := service.NewCacheService(cacheRepo)
	if err != nil {
		log.Fatalf("failed to create cache service: %v", err)
	}

	summaryService, err := service.NewSummaryService(summaryRepo, cacheService)
	if err != nil {
		log.Fatalf("failed to create summary service: %v", err)
	}

	taskService, err := service.NewTaskService(taskRepo, cacheService)
	if err != nil {
		log.Fatalf("failed to create task service: %v", err)
	}

	return summaryService, taskService, cacheService
}

func initializeRequestHandler(summaryService *service.SummaryService, taskService *service.TaskService) *handler.RequestHandler {
	requestHandler := handler.NewRequestHandler(summaryService, taskService)
	return requestHandler
}

func main() {
	config, err := config.LoadConfig()
	if err != nil {
		log.Fatalf("failed to load config: %v", err)
	}

	mongoClient, redisClient := initializeClients(config)
	summaryRepo, taskRepo, cacheRepo := initializeRepositories(mongoClient, redisClient, config)
	summaryService, taskService, _ := initializeServices(summaryRepo, taskRepo, cacheRepo)
	requestHandler := initializeRequestHandler(summaryService, taskService)

	app := gin.Default()
	app.Use(gin.Recovery())
	app.Use(gin.Logger())
	app.Use(ErrorHandlingMiddleware())

	app.GET("/health", func(c *gin.Context) {
		c.JSON(200, gin.H{
			"status": "OK",
		})
	})

	app.GET("/summary/", requestHandler.GetSummary)
	app.GET("/task/", requestHandler.GetTaskStatus)

	log.Fatal(app.Run(":8080"))
}

func ErrorHandlingMiddleware() gin.HandlerFunc {
	return func(c *gin.Context) {
		defer func() {
			if err := recover(); err != nil {
				slog.Error("CRITICAL PANIC RECOVERED",
					"error", err,
					"stack", string(debug.Stack()),
					"path", c.Request.URL.Path,
				)

				c.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{
					"code":    "INTERNAL_SERVER_ERROR",
					"message": "An unexpected error occurred.",
				})
			}
		}()

		c.Next()

		if len(c.Errors) > 0 {
			err := c.Errors.Last().Err

			switch {
			case errors.Is(err, domain.ErrVideoURLEmpty),
				errors.Is(err, domain.ErrTaskIDEmpty),
				errors.Is(err, domain.ErrDBInvalidObjectID):
				slog.Warn("Client validation error", "error", err.Error(), "path", c.Request.URL.Path)
				c.AbortWithStatusJSON(http.StatusBadRequest, gin.H{
					"code":    "BAD_REQUEST",
					"message": err.Error(), // Safe to show: "video URL is empty", etc.
				})
			case errors.Is(err, domain.ErrSummaryNotFound),
				errors.Is(err, domain.ErrTaskNotFound),
				errors.Is(err, domain.ErrTaskIDNotFound):
				slog.Info("Resource not found", "error", err.Error(), "path", c.Request.URL.Path)
				c.AbortWithStatusJSON(http.StatusNotFound, gin.H{
					"code":    "NOT_FOUND",
					"message": err.Error(), // Safe to show: "summary not found", etc.
				})
			default:
				slog.Error("Unhandled infrastructure or system error",
					"error", err.Error(),
					"path", c.Request.URL.Path,
				)

				// Mask the error completely from the client to prevent security leaks
				c.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{
					"code":    "INTERNAL_SERVER_ERROR",
					"message": "An unexpected server error occurred. Please try again later.",
				})
			}
		}
	}
}

import (
	"fmt"
	"os"
	"strconv"

	"github.com/jishnu70/Youtube_video_summarizer/internal/domain"
)

type Config struct {

	// mongo related configuration
	MongoURI      string
	MongoDatabase string

	// collection names
	TaskCollection    string
	SummaryCollection string

	// redis related configuration
	RedisHost     string
	RedisPort     int
	RedisAddr     string
	RedisPassword string
}

func LoadConfig() (*Config, error) {
	config := &Config{}
	if err := config.LoadMongoConfig(); err != nil {
		return nil, err
	}
	if err := config.loadRedisConfig(); err != nil {
		return nil, err
	}

	return config, nil
}

func (c *Config) LoadMongoConfig() error {
	envMapping := []struct {
		key   string
		value *string
	}{
		{"MONGO_URI", &c.MongoURI},
		{"MONGO_DATABASE", &c.MongoDatabase},
		{"TASK_COLLECTION", &c.TaskCollection},
		{"SUMMARY_COLLECTION", &c.SummaryCollection},
	}

	for _, env := range envMapping {
		value := os.Getenv(env.key)
		if value == "" {
			return domain.NewAppError(
				domain.ConfigLayer,
				domain.CodeInvalidInput,
				fmt.Sprintf("%s: %s", domain.ErrConfigMissingField, env.key),
				nil,
			)
		}
		*env.value = value
	}
	return nil
}

func (c *Config) loadRedisConfig() error {
	envMapping := []struct {
		key   string
		value *string
	}{
		{"REDIS_HOST", &c.RedisHost},
		{"REDIS_PASSWORD", &c.RedisPassword},
	}

	for _, env := range envMapping {
		value := os.Getenv(env.key)
		if value == "" {
			return domain.NewAppError(
				domain.ConfigLayer,
				domain.CodeInvalidInput,
				fmt.Sprintf("%s: %s", domain.ErrConfigMissingField, env.key),
				nil,
			)
		}
		*env.value = value
	}

	// Handle REDIS_PORT separately since it needs to be converted to an integer
	portStr := os.Getenv("REDIS_PORT")
	if portStr == "" {
		return domain.NewAppError(
			domain.ConfigLayer,
			domain.CodeInvalidInput,
			fmt.Sprintf("%s: %s", domain.ErrConfigMissingField, "REDIS_PORT"),
			nil,
		)
	}

	port, err := strconv.Atoi(portStr)
	if err != nil {
		return domain.NewAppError(
			domain.ConfigLayer,
			domain.CodeInvalidInput,
			fmt.Sprintf("%s: %s", domain.ErrConfigInvalidValue, "REDIS_PORT"),
			err,
		)
	}
	c.RedisPort = port

	c.RedisAddr = fmt.Sprintf("%s:%d", c.RedisHost, c.RedisPort)

	return nil
}

// - video_id of the video
// - video_url of the video

// - created_at timestamp of when the summary was created

	VideoURL  string        `bson:"video_url" json:"video_url"`

		ID:        s.ID.Hex(),

		Summary:   s.Summary,
		ModelName: s.ModelName,

/*
 * DB struct representing the BackgroundTaskStatus entity in the database.
 * BackgroundTaskStatusDB is a struct for a background task status:
 * 	TaskID: unique identifier for the task

	Message   string            `bson:"message" json:"message"`
	Status    domain.TaskStatus `bson:"status" json:"status"`
	CreatedAt time.Time         `bson:"created_at" json:"created_at"`
	UpdatedAt time.Time         `bson:"updated_at" json:"updated_at"`
}

	return &domain.BackgroundTaskStatus{
		ID:        b.ID.Hex(),

		Message:   b.Message,
		Status:    b.Status,
		CreatedAt: b.CreatedAt,
		UpdatedAt: b.UpdatedAt,
	}
}

import (
	"context"
	"errors"
	"time"

	"github.com/jishnu70/Youtube_video_summarizer/internal/domain"
	"go.mongodb.org/mongo-driver/v2/mongo"
	"go.mongodb.org/mongo-driver/v2/mongo/options"
)

type MongoDB struct {
	// Add fields for MongoDB connection, e.g., client, database, collection, etc.
	client   *mongo.Client
	database *mongo.Database
}

func New(uri, databaseName string) (*MongoDB, error) {
	// Initialize MongoDB connection here using the provided URI
	serverAPI := options.ServerAPI(options.ServerAPIVersion1)
	clientOptions := options.Client().ApplyURI(uri).SetServerAPIOptions(serverAPI).SetTimeout(5 * time.Second)

	client, err := mongo.Connect(clientOptions)
	if err != nil {
		if errors.Is(err, context.DeadlineExceeded) {
			return nil, domain.NewAppError(
				domain.DatabaseLayer,
				domain.CodeTimeout,
				domain.ErrDBConnectionTimeout,
				err,
			)
		}
		return nil, domain.NewAppError(
			domain.DatabaseLayer,
			domain.CodeConnectionError,
			domain.ErrDBConnectionFailed,
			err,
		)
	}

	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	if pingErr := client.Ping(ctx, nil); pingErr != nil {
		_ = client.Disconnect(ctx)
		return nil, domain.NewAppError(
			domain.DatabaseLayer,
			domain.CodeConnectionError,
			domain.ErrDBConnectionFailed,
			pingErr,
		)
	}
	database := client.Database(databaseName)

	return &MongoDB{
		client:   client,
		database: database,
	}, nil
}

func (m *MongoDB) Ping(ctx context.Context) error {
	// Implement a ping method to check the connection to MongoDB
	if err := m.client.Ping(ctx, nil); err != nil {
		if errors.Is(err, context.DeadlineExceeded) {
			return domain.NewAppError(
				domain.DatabaseLayer,
				domain.CodeTimeout,
				domain.ErrDBConnectionTimeout,
				err,
			)
		}
		return domain.NewAppError(
			domain.DatabaseLayer,
			domain.CodeConnectionError,
			domain.ErrDBPingFailed,
			err,
		)
	}
	return nil
}

func (m *MongoDB) Close(ctx context.Context) error {
	// Implement a method to close the MongoDB connection
	if err := m.client.Disconnect(ctx); err != nil {
		return domain.NewAppError(
			domain.DatabaseLayer,
			domain.CodeConnectionError,
			domain.ErrDBDisconnectFailed,
			err,
		)
	}
	return nil
}

func (m *MongoDB) Collection(collectionName string) *mongo.Collection {
	// Implement a method to get a collection from the MongoDB database
	return m.database.Collection(collectionName)
}

	case TaskStatusQueued,

		TaskStatusCompleted,
		TaskStatusFailed,
		TaskStatusTimedOut,
		TaskStatusCancelled,
		TaskStatusUnknown,
		TaskStatusStarted,
		TaskStatusTranscribing:
		return true
	default:
		return false
	}
}

type AppError struct {
	ErrLayer ErrorLayer
	ErrCode  ErrorCode
	Message  string
	Err      error
}

func (e AppError) Error() string {
	if e.Err != nil {
		return fmt.Sprintf("[%s][%s] %s: %v", e.ErrLayer, e.ErrCode, e.Message, e.Err)
	}
	return fmt.Sprintf("[%s][%s] %s", e.ErrLayer, e.ErrCode, e.Message)
}

func (e AppError) Unwrap() error {
	return e.Err
}

func NewAppError(layer ErrorLayer, code ErrorCode, message string, err error) error {
	return &AppError{
		ErrLayer: layer,
		ErrCode:  code,
		Message:  message,
		Err:      err,
	}
}

// api/internal/repo/cache.go

import (
	"context"
	"errors"
	"fmt"
	"time"

	domain "github.com/jishnu70/Youtube_video_summarizer/internal/domain"
	redis "github.com/redis/go-redis/v9"
)

type CacheRepo struct {
	client *redis.Client
}

func NewCacheRepo(redisClient *redis.Client) *CacheRepo {
	return &CacheRepo{
		client: redisClient,
	}
}

type method string

const (
	GET method = "GET"
	SET method = "SET"
	DEL method = "DEL"
)

var metToStr = map[method]string{
	GET: domain.ErrCacheGetFailed,
	SET: domain.ErrCacheSetFailed,
	DEL: domain.ErrCacheDeleteFailed,
}

func handleRedisError(err error, met method) error {
	if errors.Is(err, redis.ErrPoolTimeout) {
		return domain.NewAppError(
			domain.CacheLayer,
			domain.ErrCacheConnectionTimeout,
			fmt.Sprintf("CacheRepo: %s", domain.ErrCacheConnectionTimeout),
			err,
		)
	}
	if errors.Is(err, redis.ErrClosed) {
		return domain.NewAppError(
			domain.CacheLayer,
			domain.ErrCacheConnectionFailed,
			fmt.Sprintf("CacheRepo: %s", domain.ErrCacheConnectionFailed),
			err,
		)
	}
	if errors.Is(err, redis.Nil) {
		return domain.NewAppError(
			domain.CacheLayer,
			domain.ErrCacheKeyNotFound,
			fmt.Sprintf("CacheRepo: %s", domain.ErrCacheKeyNotFound),
			err,
		)
	}
	return domain.NewAppError(
		domain.CacheLayer,
		domain.ErrCacheGetFailed,
		fmt.Sprintf("CacheRepo: %s", metToStr[met]),
		err,
	)
}

func (r *CacheRepo) Set(ctx context.Context, key string, value any, ttl int64) error {
	duration := time.Duration(ttl) * time.Second
	if err := r.client.Set(ctx, key, value, duration).Err(); err != nil {
		return handleRedisError(err, SET)
	}
	return nil
}

func (r *CacheRepo) SetNx(ctx context.Context, key string, value any, ttl int64) (bool, error) {
	duration := time.Duration(ttl) * time.Second
	isSet, err := r.client.SetNX(ctx, key, value, duration).Result()
	// isSet will be true if the key was set, false if it already exists
	if err != nil {
		return false, handleRedisError(err, SET)
	}
	return isSet, nil
}

func (r *CacheRepo) Get(ctx context.Context, key string) ([]byte, error) {
	value, err := r.client.Get(ctx, key).Bytes()
	if err != nil {
		return nil, handleRedisError(err, GET)
	}
	return value, nil
}

func (r *CacheRepo) GetTTl(ctx context.Context, key string) (time.Duration, error) {
	ttl, err := r.client.TTL(ctx, key).Result()
	if err != nil {
		return 0, handleRedisError(err, GET)
	}
	return ttl, nil
}

func (r *CacheRepo) Exists(ctx context.Context, key string) (bool, error) {
	exists, err := r.client.Exists(ctx, key).Result()
	if err != nil {
		return false, handleRedisError(err, GET)
	}
	return exists > 0, nil
}

func (r *CacheRepo) Delete(ctx context.Context, key string) error {
	if err := r.client.Del(ctx, key).Err(); err != nil {
		return handleRedisError(err, DEL)
	}
	return nil
}

// Common Repository error handlers

func handleRepoFindError(err error, repoName string) error {
	if mongo.IsTimeout(err) {
		return domain.NewAppError(
			domain.RepoLayer,
			domain.CodeTimeout,
			fmt.Sprintf("%s: %s", repoName, domain.ErrDBConnectionTimeout),
			err,
		)
	}
	if errors.Is(err, mongo.ErrClientDisconnected) {
		return domain.NewAppError(
			domain.RepoLayer,
			domain.CodeConnectionError,
			fmt.Sprintf("%s: %s", repoName, domain.ErrDBConnectionFailed),
			err,
		)
	}
	if errors.Is(err, mongo.ErrNoDocuments) {
		return domain.NewAppError(
			domain.RepoLayer,
			domain.CodeNotFound,
			fmt.Sprintf("%s: %s", repoName, domain.ErrDBNoDocuments),
			err,
		)
	}
	return domain.NewAppError(
		domain.RepoLayer,
		domain.CodeInternalError,
		fmt.Sprintf("%s: %s", repoName, domain.ErrDBFindFailed),
		err,
	)
}

func handleRepoInsertError(err error, repoName string) error {
	if mongo.IsDuplicateKeyError(err) {
		return domain.NewAppError(
			domain.RepoLayer,
			domain.CodeAlreadyExists,
			fmt.Sprintf("%s: %s", repoName, domain.ErrTaskAlreadyExists),
			err,
		)
	}
	if mongo.IsTimeout(err) {
		return domain.NewAppError(
			domain.RepoLayer,
			domain.CodeTimeout,
			fmt.Sprintf("%s: %s", repoName, domain.ErrDBConnectionTimeout),
			err,
		)
	}
	if errors.Is(err, mongo.ErrClientDisconnected) {
		return domain.NewAppError(
			domain.RepoLayer,
			domain.CodeConnectionError,
			fmt.Sprintf("%s: %s", repoName, domain.ErrDBConnectionFailed),
			err,
		)
	}
	return domain.NewAppError(
		domain.RepoLayer,
		domain.CodeInternalError,
		fmt.Sprintf("%s: %s", repoName, domain.ErrDBInsertFailed),
		err,
	)
}

func handleRepoDeleteError(err error, repoName string) error {
	if mongo.IsTimeout(err) {
		return domain.NewAppError(
			domain.RepoLayer,
			domain.CodeTimeout,
			fmt.Sprintf("%s: %s", repoName, domain.ErrDBConnectionTimeout),
			err,
		)
	}
	if errors.Is(err, mongo.ErrClientDisconnected) {
		return domain.NewAppError(
			domain.RepoLayer,
			domain.CodeConnectionError,
			fmt.Sprintf("%s: %s", repoName, domain.ErrDBConnectionFailed),
			err,
		)
	}
	if errors.Is(err, mongo.ErrNoDocuments) {
		return domain.NewAppError(
			domain.RepoLayer,

		domain.RepoLayer,
		domain.CodeInternalError,

// api/internal/service/cache.go

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"strings"

	"github.com/jishnu70/Youtube_video_summarizer/internal/domain"
	repo "github.com/jishnu70/Youtube_video_summarizer/internal/repo"
)

type CacheService struct {
	cacheRepo *repo.CacheRepo
}

func NewCacheService(cacheRepo *repo.CacheRepo) (*CacheService, error) {
	if cacheRepo == nil {
		return nil, domain.ErrCacheRepoNotInitialized
	}
	return &CacheService{
		cacheRepo: cacheRepo,
	}, nil
}

type cacheType string

const (
	SummaryCacheKeyPrefix  cacheType = "summary:"
	CacheTaskIDPrefix      cacheType = "task:id:"
	CacheURLToTaskIDPrefix cacheType = "task:url:"
)

func generateCacheKey(videoURL string, keyType cacheType) string {
	return string(keyType) + videoURL
}

func structToString(value any) ([]byte, error) {
	jsonBytes, err := json.Marshal(value)
	if err != nil {
		return nil, err
	}
	return jsonBytes, nil
}

func stringToStruct(data []byte, value any) error {
	if err := json.Unmarshal(data, value); err != nil {
		return err
	}
	return nil
}

func (s *CacheService) setCache(ctx context.Context, key string, value any, keyType cacheType, ttl int64) error {
	if strings.TrimSpace(key) == "" {
		return errors.New("cache key is empty")
	}
	if value == nil {
		return errors.New("cache value is nil")
	}
	payload, err := structToString(value)
	if err != nil {
		return err
	}
	return s.cacheRepo.Set(ctx, generateCacheKey(key, keyType), payload, ttl)
}

func (s *CacheService) getCache(ctx context.Context, key string, value any, keyType cacheType) error {
	if strings.TrimSpace(key) == "" {
		return errors.New("cache key is empty")
	}
	data, err := s.cacheRepo.Get(ctx, generateCacheKey(key, keyType))
	if err != nil {
		return err
	}
	return stringToStruct(data, value)
}

func (s *CacheService) CacheSummary(ctx context.Context, videoURL string, value *domain.Summary, ttl int64) error {
	if err := s.setCache(ctx, videoURL, value, SummaryCacheKeyPrefix, ttl); err != nil {
		if errors.Is(err, domain.ErrCacheConnectionFailed) {
			return fmt.Errorf("failed to cache summary due to cache connection failure: %w", err)
		}
		if errors.Is(err, domain.ErrCacheSetFailed) {
			return fmt.Errorf("failed to cache summary due to cache set failure: %w", err)
		}
		if errors.Is(err, domain.ErrCacheConnectionTimeout) {
			return fmt.Errorf("failed to cache summary due to cache connection timeout: %w", err)
		}
		return fmt.Errorf("failed to cache summary: %w", err)
	}
	return nil
}

func (s *CacheService) CacheTask(ctx context.Context, taskID string, value *domain.BackgroundTaskStatus, ttl int64) error {
	if err := s.setCache(ctx, taskID, value, CacheTaskIDPrefix, ttl); err != nil {
		if errors.Is(err, domain.ErrCacheConnectionFailed) {
			return fmt.Errorf("failed to cache task due to cache connection failure: %w", err)
		}
		if errors.Is(err, domain.ErrCacheSetFailed) {
			return fmt.Errorf("failed to cache task due to cache set failure: %w", err)
		}
		if errors.Is(err, domain.ErrCacheConnectionTimeout) {
			return fmt.Errorf("failed to cache task due to cache connection timeout: %w", err)
		}
		return fmt.Errorf("failed to cache task: %w", err)
	}
	return nil
}

func (r *CacheService) CacheVideoURLToTaskID(ctx context.Context, videoURL string, taskID string, ttl int64) error {
	if strings.TrimSpace(videoURL) == "" {
		return domain.ErrVideoURLEmpty
	}
	if strings.TrimSpace(taskID) == "" {
		return domain.ErrTaskIDEmpty
	}
	if err := r.cacheRepo.Set(ctx, generateCacheKey(videoURL, CacheURLToTaskIDPrefix), taskID, ttl); err != nil {
		if errors.Is(err, domain.ErrCacheConnectionFailed) {
			return fmt.Errorf("failed to cache videoURL to taskID mapping due to cache connection failure: %w", err)
		}
		if errors.Is(err, domain.ErrCacheSetFailed) {
			return fmt.Errorf("failed to cache videoURL to taskID mapping due to cache set failure: %w", err)
		}
		if errors.Is(err, domain.ErrCacheConnectionTimeout) {
			return fmt.Errorf("failed to cache videoURL to taskID mapping due to cache connection timeout: %w", err)
		}
		return fmt.Errorf("failed to cache videoURL to taskID mapping: %w", err)
	}
	return nil
}

func (s *CacheService) GetCachedSummary(ctx context.Context, videoURL string) (*domain.Summary, error) {
	var summary domain.Summary
	if err := s.getCache(ctx, videoURL, &summary, SummaryCacheKeyPrefix); err != nil {
		if errors.Is(err, domain.ErrCacheConnectionFailed) {
			return nil, fmt.Errorf("failed to get cached summary due to cache connection failure: %w", err)
		}
		if errors.Is(err, domain.ErrCacheKeyNotFound) {
			return nil, domain.ErrSummaryNotFound
		}
		if errors.Is(err, domain.ErrCacheConnectionTimeout) {
			return nil, fmt.Errorf("failed to get cached summary due to cache connection timeout: %w", err)
		}
		return nil, err
	}
	return &summary, nil
}

func (s *CacheService) GetCachedTaskIDByVideoURL(ctx context.Context, videoURL string) (string, error) {
	if strings.TrimSpace(videoURL) == "" {
		return "", domain.ErrVideoURLEmpty
	}
	taskID, err := s.cacheRepo.Get(ctx, generateCacheKey(videoURL, CacheURLToTaskIDPrefix))
	if err != nil {
		if errors.Is(err, domain.ErrCacheConnectionFailed) {
			return "", fmt.Errorf("failed to get cached taskID due to cache connection failure: %w", err)
		}
		if errors.Is(err, domain.ErrCacheKeyNotFound) {
			return "", domain.ErrTaskIDNotFound
		}
		if errors.Is(err, domain.ErrCacheConnectionTimeout) {
			return "", fmt.Errorf("failed to get cached taskID due to cache connection timeout: %w", err)
		}
		return "", fmt.Errorf("failed to get cached taskID: %w", err)
	}
	return string(taskID), nil
}

func (s *CacheService) GetCachedTaskStatus(ctx context.Context, taskID string) (*domain.BackgroundTaskStatus, error) {
	var taskStatus domain.BackgroundTaskStatus
	if err := s.getCache(ctx, taskID, &taskStatus, CacheTaskIDPrefix); err != nil {
		if errors.Is(err, domain.ErrCacheConnectionFailed) {
			return nil, fmt.Errorf("failed to get cached task status due to cache connection failure: %w", err)
		}
		if errors.Is(err, domain.ErrCacheKeyNotFound) {
			return nil, domain.ErrTaskNotFound
		}
		if errors.Is(err, domain.ErrCacheConnectionTimeout) {
			return nil, fmt.Errorf("failed to get cached task status due to cache connection timeout: %w", err)
		}
		return nil, fmt.Errorf("failed to get cached task status: %w", err)
	}
	return &taskStatus, nil
}

func (s *TaskService) CreateTask(ctx context.Context, videoURL string) (*domain.BackgroundTaskStatus, error) {
	videoURL = strings.TrimSpace(videoURL)
	if videoURL == "" {
		return nil, domain.ErrVideoURLEmpty
	}
	now := time.Now().UTC()
	task := &domain.BackgroundTaskStatus{
		VideoURL:  videoURL,
		Message:   "The video is queued for processing",
		Status:    domain.TaskStatusQueued,
		CreatedAt: now,
		UpdatedAt: now,
	}
	result, err := s.taskRepo.Insert(ctx, task)
	if err != nil {
		return nil, fmt.Errorf("failed to create task: %w", err)
	}
	err = s.cacheService.CacheTask(ctx, result.ID, result, 3600) // Cache for 1 hour
	if err != nil {
		return nil, fmt.Errorf("failed to cache task: %w", err)
	}
	err = s.cacheService.CacheVideoURLToTaskID(ctx, videoURL, result.ID, 3600) // Cache for 1 hou
	if err != nil {
		return nil, fmt.Errorf("failed to cache videoURL to taskID mapping: %w", err)
	}
	return result, nil
}

func (s *TaskService) GetTaskStatusByTaskID(ctx context.Context, taskID string) (*domain.BackgroundTaskStatus, error) {
	taskID = strings.TrimSpace(taskID)
	if taskID == "" {
		return nil, domain.ErrTaskIDEmpty
	}
	task, err := s.cacheService.GetCachedTaskStatus(ctx, taskID)
	if err == nil {
		return task, nil
	}

	if err != nil {
		return nil, fmt.Errorf("failed to fetch task from database: %w", err)
	}
	err = s.cacheService.CacheTask(ctx, task.ID, task, 120) // Cache for 2 minutes
	if err != nil {

	}
	return task, nil
}

func (s *TaskService) GetTaskStatusByVideoURL(ctx context.Context, videoURL string) (*domain.BackgroundTaskStatus, error) {

	if videoURL == "" {
		return nil, domain.ErrVideoURLEmpty
	}
	taskID, err := s.cacheService.GetCachedTaskIDByVideoURL(ctx, videoURL)
	if err == nil {
		result, err := s.cacheService.GetCachedTaskStatus(ctx, taskID)
		if err != nil {
			return s.GetTaskStatusByTaskID(ctx, taskID)
		}
		return result, nil
	}
	result, err := s.taskRepo.FindByVideoURL(ctx, videoURL)
	if err != nil {
		return nil, fmt.Errorf("failed to fetch task from database: %w", err)
	}
	err = s.cacheService.CacheTask(ctx, result.ID, result, 120) // Cache for 2 minutes
	if err != nil {

	}
	err = s.cacheService.CacheVideoURLToTaskID(ctx, videoURL, result.ID, 3600) // Cache for 1 hour

	}

type RequestHandler struct {

func NewRequestHandler(
	summaryService *service.SummaryService,
	taskService *service.TaskService,
) *RequestHandler {
	if summaryService == nil {
		panic("summaryService is nil")
	}
	if taskService == nil {
		panic("taskService is nil")
	}
	return &RequestHandler{
		summaryService: summaryService,
		taskService:    taskService,
	}

func (h *RequestHandler) GetSummary(ctx *gin.Context) {
	var videoURLBinding schema.SummaryRequest
	if err := ctx.ShouldBindQuery(&videoURLBinding); err != nil {
		ctx.Error(err)
		return
	}

	summary, err := h.summaryService.GetSummaryByVideoURL(ctx.Request.Context(), videoURLBinding.VideoURL)
	if err != nil {
		if errors.Is(err, domain.ErrSummaryNotFound) {
			if exists, checkErr := checkYouTubeVideoExists(videoURLBinding.VideoURL); checkErr != nil {
				ctx.Error(checkErr)
				return
			} else if !exists {
				ctx.JSON(http.StatusNotFound, gin.H{"error": "YouTube video does not exist."})
				return
			}

			// Create a new background task for the video
			task, taskErr := h.taskService.CreateTask(ctx.Request.Context(), videoURLBinding.VideoURL)
			if taskErr != nil {
				ctx.Error(fmt.Errorf("failed to create background task: %w", taskErr))
				return
			}
			ctx.JSON(http.StatusAccepted, schema.NewTaskStatusResponse(task))
			return
		}
		ctx.Error(err)
		return
	}

	ctx.JSON(http.StatusOK, schema.NewSummaryResponse(summary))
}

func (h *RequestHandler) GetTaskStatus(ctx *gin.Context) {
	var taskIDBinding schema.TaskStatusRequest
	if err := ctx.ShouldBindQuery(&taskIDBinding); err != nil {
		ctx.JSON(400, gin.H{"error": err.Error()})
		return
	}

	taskID := taskIDBinding.TaskID
	videoURL := taskIDBinding.VideoURL
	if taskID == "" && videoURL == "" {
		ctx.JSON(http.StatusBadRequest, gin.H{"error": "Invalid request: insufficient parameters provided."})
		return
	}

	var taskStatus *domain.BackgroundTaskStatus
	var err error
	if taskID != "" {
		taskStatus, err = h.taskService.GetTaskStatusByTaskID(ctx.Request.Context(), taskID)
	} else {
		taskStatus, err = h.taskService.GetTaskStatusByVideoURL(ctx.Request.Context(), videoURL)
	}

	if err != nil {
		ctx.Error(err)
		return
	}

	if taskStatus == nil {
		ctx.Error(domain.ErrTaskNotFound)
		return
	}

	ctx.JSON(http.StatusOK, schema.NewTaskStatusResponse(taskStatus))
}

func checkYouTubeVideoExists(videoURL string) (bool, error) {
	// Build the oEmbed endpoint URL

	_, err := url.ParseRequestURI(videoURL)
	if err != nil {
		return false, fmt.Errorf("invalid video URL: %w", err)
	}

	oembedBaseURL := "https://www.youtube.com/oembed"
	params := url.Values{}
	params.Set("url", videoURL)
	oembedURL := fmt.Sprintf("%s?%s", oembedBaseURL, params.Encode())

	client := &http.Client{Timeout: 5 * time.Second}
	resp, err := client.Get(oembedURL)
	if err != nil {
		return false, err
	}
	defer resp.Body.Close()

	// If the video doesn't exist, is private, or is deleted,
	// the oEmbed API reliably returns a 404 Not Found or 400 Bad Request.
	if resp.StatusCode == http.StatusOK {
		return true, nil
	}

	return false, nil
}
